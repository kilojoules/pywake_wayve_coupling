69 def gauss_alternative(turbines,Uinf,Vinf,kwake):
70  '''
71  Alternative implementation of Gaussian wake model (depreciated)
72 
73  Sort wind turbines and then advance through farm. Is faster than building
74  a general matrix, but issues arise with the Jacobian when the order of turbines
75  changes due to the perturbation
76  '''
77  Nt = len(turbines) #Number of turbines
78  Nq = 16 #Number of quadrature points
79  Sinf = np.sqrt(Uinf**2+Vinf**2) #Wind speed
80  e_str = np.array([Uinf,Vinf])/Sinf #Unit vector along the wind direction
81  e_span = np.array([-Vinf,Uinf])/Sinf #Unit vector in cross wind direction
82  #Sort turbines along wind direction
83  xs = np.array([turbines[i].x for i in range(Nt)])
84  ys = np.array([turbines[i].y for i in range(Nt)])
85  coordinates = np.concatenate([xs,ys]).reshape(Nt,2,order='F')
86  dist = np.dot(coordinates,e_str)
87  order = np.argsort(dist)
88  turbines_sort = [turbines[i] for i in order]
89  #Compute velocities
90  S = np.zeros((Nt))
91  S[0] = Sinf
92  for i in range(1,Nt):
93  turbi = turbines_sort[i]
94  wake_deficit = 0.
95  for k in range(0,i):
96  turbk = turbines_sort[k]
97  #Loop over quadrature points
98  for n in range(Nq):
99  wn = 1.0/Nq
100  phi = 2*np.pi*n/Nq
101  r = np.sqrt((3+(1-2*(n%2))*np.sqrt(3))/6)*turbi.D/2.0
102  xn = turbi.x + r*np.cos(phi)*e_span[0]
103  yn = turbi.y + r*np.cos(phi)*e_span[1]
104  zn = r*np.sin(phi) #Assuming all turbines have same zh
105  #Vector from turbine K to point n on turbine I
106  KI = np.array([xn-turbk.x,yn-turbk.y])
107  #Streamwise distance between turbine K and turbine I along e_str
108  #Positive if I is downstream of K
109  #Negative if I is upstream of K
110  delta_str = np.dot(KI,e_str)
111  #Spanwise distance between turbine K and turbine I along e_span
112  #Has a sign but wake model is axisymmetric
113  delta_span = np.dot(KI,e_span)
114  #Compute wakedeficit at turbine I due to turbine K
115  if delta_str>turbk.D:
116  wake_deficit += S[k]*wn*(1-gauss_wakedeficit(delta_str,
117  delta_span,zn,
118  turbk.D,turbk.Ct,
119  kwake) )
120  #Ghost turbine?
121  S[i] = Sinf - wake_deficit
122  #Unsort turbines
123  inv_order = np.argsort(order)
124  return S[inv_order]

126 def gauss_TI(turbines,abl):
127  '''
128  Turbulence intensity model of Niayifar and Porte-Agel (2016)
129 
130  Turbulence intensity is assumed not to depend on the perturbation velocities so
131  that, contrary to the Gaussian wake model itself, the turbulence intensity model
132  can be calculated by ordering the turbines and then sweeping through the farm
133 
134  Parameters
135  ----------
136  turbines: list of turbine objects (defined in TLMForcing.py)
137  list with the wind turbines
138  abl: ABL object (defined in TLMForcing.py)
139  atmospheric state
140 
141  Returns
142  -------
143  TI: 1d numpy array
144  local turbulent intensity at the turbines
145  '''
146  Nt = len(turbines) #Number of turbines
147  Uinf = abl.U1
148  Vinf = abl.V1
149  e_str = e_streamwise(Uinf,Vinf) #Unit vector along the wind direction
150  e_span = e_spanwise(Uinf,Vinf) #Unit vector in cross wind direction
151  TI = np.zeros((Nt))
152  #Sort turbines along wind direction
153  xs = np.array([turbines[i].x for i in range(Nt)])
154  ys = np.array([turbines[i].y for i in range(Nt)])
155  coordinates = np.concatenate([xs,ys]).reshape(Nt,2,order='F')
156  dist = np.dot(coordinates,e_str)
157  order = np.argsort(dist)
158  turbines_sort = [turbines[i] for i in order]
159  #Compute turbulent intensities
160  TI[0] = abl.TI
161  for i in range(1,Nt):
162  turbi = turbines_sort[i]
163  TIadded = np.zeros((i))
164  #Compute added streamwise turbulent intensity induced by turbine k
165  #at turbine i
166  for k in range(0,i):
167  kwake = abl.kwake(TI[k])
168  turbk = turbines_sort[k]
169  #Vector from turbine K to point n on turbine I
170  KI = np.array([turbi.x-turbk.x,turbi.y-turbk.y])
171  #Streamwise distance between turbine K and turbine I along e_str
172  #Positive if I is downstream of K
173  #Negative if I is upstream of K
174  delta_str = np.dot(KI,e_str)
175  #Spanwise distance between turbine K and turbine I along e_span
176  #Has a sign but wake model is axisymmetric
177  delta_span = np.dot(KI,e_span)
178  #Turbine k lies upstream of Turbine i
179  if delta_str>0:# and delta_str<15*turbk.D:
180  beta = 0.5*(1+np.sqrt(1-turbk.Ct))/np.sqrt(1-turbk.Ct)
181  eps = 0.2*np.sqrt(beta)
182  sigma = kwake*delta_str+turbk.D*eps
183  rwake = 2*sigma
184  rdisk = turbi.D/2.0
185  #TI wake of k intersects with rotor i
186  if np.abs(delta_span)>=(rwake+rdisk):
187  Aw = 0. #No overlap
188  elif np.abs(delta_span)>=max([rwake,rdisk]):
189  x = (delta_span**2-rdisk**2+rwake**2)/(2*np.abs(delta_span))
190  Aw = ( area_circle_segment(rwake,x)
191  +area_circle_segment(rdisk,np.abs(delta_span)-x) )
192  elif np.abs(delta_span)>=np.abs(rwake-rdisk):
193  R = max([rwake,rdisk])
194  r = min([rwake,rdisk])
195  x = (delta_span**2-r**2+R**2)/(2*np.abs(delta_span))
196  Aw = ( area_circle_reflex(r,x-np.abs(delta_span))
197  +area_circle_segment(R,x) )
198  else:
199  Aw = np.pi*min([rwake,rdisk])**2
200 
201  induction = (1-np.sqrt(1-turbk.Ct))/2.
202  Iadded = (0.73 * induction**(0.8325)
203  * abl.TI**(0.0325)
204  * (delta_str/turbk.D)**(-0.32) )
205  TIadded[k] = Aw/(np.pi*rdisk**2)*Iadded
206  TI[i] = np.sqrt(abl.TI**2 + (np.max(TIadded))**2)
207 
208  #Unsort turbines
209  inv_order = np.argsort(order)
210  return TI[inv_order]

213  '''
214  Compute system matrix A of Gaussian wake model
215 
216  Parameters
217  ----------
218  turbines: list of turbine objects (defined in TLMForcing.py)
219  list with the wind turbines
220  abl: ABL object (defined in TLMForcing.py)
221  atmospheric state
222  TI: 1d numpy array
223  local turbulent intensity at the turbines
224 
225  Returns
226  -------
227  A: numpy array with shape (Nt,Nt)
228  system matrix of Gaussian wake model
229  '''
230  Nt = len(turbines) #Number of turbines
231  Uinf = abl.U1
232  Vinf = abl.V1
233  Nq = 16 #Number of quadrature pints
234  e_str = e_streamwise(Uinf,Vinf) #Unit vector along the wind direction
235  e_span = e_spanwise(Uinf,Vinf) #Unit vector in cross wind direction
236  A = np.identity((Nt))
237  for i, turbi in enumerate(turbines):
238  #Loop over all other turbines to collect wake effects
239  for k, turbk in enumerate(turbines):
240  kwake = abl.kwake(TI[k])
241  #minD = minimum distance between turbines to have wake interaction
242  #(if x<minD, the wake deficit is not defined)
243  beta = 0.5*(1+np.sqrt(1-turbk.Ct))/np.sqrt(1-turbk.Ct)
244  eps = 0.2*np.sqrt(beta)
245  minD = 1./kwake*(np.sqrt(turbk.Ct/8.)-eps)*turbk.D
246  #Vector from turbine K to center of turbine I
247  KI = np.array([turbi.x-turbk.x,turbi.y-turbk.y])
248  #Streamwise distance between turbine K and turbine I along e_str
249  #Positive if I is downstream of K
250  #Negative if I is upstream of K
251  delta_str = np.dot(KI,e_str)
252  #test distance between center of turbine I and K.
253  #If yaw angle of both turbines is equal, delta_str is identical for all
254  #quadrature points.
255  if delta_str<=np.max([0.,minD]): continue
256  #Loop over quadrature points
257  for n in range(Nq):
258  wn = 1.0/Nq
259  phi = 2*np.pi*n/Nq
260  r = np.sqrt((3.+(1.-2*(n%2))*np.sqrt(3.))/6.)*turbi.D/2.0
261  xn = turbi.x + r*np.cos(phi)*e_span[0]
262  yn = turbi.y + r*np.cos(phi)*e_span[1]
263  zn = r*np.sin(phi) #Assuming all turbines have same zh
264  #Vector from turbine K to point n on turbine I
265  KI = np.array([xn-turbk.x,yn-turbk.y])
266  #Streamwise distance between turbine K and turbine I along e_str
267  #Positive if I is downstream of K
268  #Negative if I is upstream of K
269  delta_str = np.dot(KI,e_str)
270  #Spanwise distance between turbine K and turbine I along e_span
271  #Has a sign but wake model is axisymmetric
272  delta_span = np.dot(KI,e_span)
273  #Compute wakedeficit at turbine I due to turbine K
274  A[i,k] += wn*gauss_wakedeficit(delta_str,delta_span,zn,
275  turbk.D,turbk.Ct,kwake)
276  #Ghost turbine?
277  return A

356 def gauss_wakedeficit(x,y,z,d0,Ct,kwake):
357  '''
358  Wake deficit function
359 
360  Parameters
361  ----------
362  x,y,z: float
363  relative streamwise, spanwise and vertical distance in the wake
364  d0: float
365  rotor diameter
366  Ct: float
367  thrust coefficient
368  kwake: float
369  wake expansion coefficient
370 
371  Returns
372  -------
373  deficit: float
374  velocity deficit
375  '''
376  beta = 0.5*(1+np.sqrt(1-Ct))/np.sqrt(1-Ct)
377  eps = 0.2*np.sqrt(beta)
378  sigma_d0 = kwake*x/d0+eps
379  deficit = ( (1-np.sqrt(1-Ct/(8*sigma_d0**2)))
380  *np.exp(-1./(2*sigma_d0**2)*( (z/d0)**2+(y/d0)**2) ) )
381  return deficit

383 def gauss_wakedeficit_jac(x,y,z,d0,Ct,kwake):
384  '''
385  Jacobian of wake deficit function
386 
387  Parameters
388  ----------
389  x,y,z: float
390  relative streamwise, spanwise and vertical distance in the wake
391  d0: float
392  rotor diameter
393  Ct: float
394  thrust coefficient
395  kwake: float
396  wake expansion coefficient
397 
398  Returns
399  -------
400  _: numpy array with size (2,)
401  Jacobian of velocity deficit w.r.t. x and y
402  '''
403  beta = 0.5*(1+np.sqrt(1-Ct))/np.sqrt(1-Ct)
404  eps = 0.2*np.sqrt(beta)
405  sigma_d0 = kwake*x/d0+eps
406  defdx = (-Ct*kwake/(8*d0*sigma_d0**3)/np.sqrt(1-Ct/(8*sigma_d0**2))*
407  np.exp(-1./(2*sigma_d0**2)*( (z/d0)**2+(y/d0)**2) ) )

409  kwake/(d0*sigma_d0**3)*( (z/d0)**2+(y/d0)**2) )
410  defdy = -gauss_wakedeficit(x,y,z,d0,Ct,kwake)*y/(d0**2*sigma_d0**2)

421 def jensen(turbines,Uinf,Vinf,kwake):
422  '''Jensen wake model (depreciated)'''
423  Sinf = np.sqrt(Uinf**2+Vinf**2) #Wind speed
424  e_str = np.array([Uinf,Vinf])/Sinf #Unit vector along the wind direction
425  e_span = np.array([-Vinf,Uinf])/Sinf #Unit vector in cross wind direction
426  S = np.zeros((len(turbines)))
427  Ftu = np.zeros((len(turbines)))
428  Ftv = np.zeros((len(turbines)))
429  for i, turbi in enumerate(turbines):
430  #Compute superposition of wake deficits
431  wakedeficit = 0.
432  for turbk in turbines:
433  #Vector from turbine K to turbine I
434  KI = np.array([turbi.x-turbk.x,turbi.y-turbk.y])
435  #Streamwise distance between turbine K and turbine I along e_str
436  #Positive if I is downstream of K
437  #Negative if I is upstream of K
438  delta_str = np.dot(KI,e_str)
439  #Spanwise distance between turbine K and turbine I along e_span
440  #Has a sign but wake model is axisymmetric
441  delta_span = np.dot(KI,e_span)
442  #Compute wakedeficit at turbine I due to turbine K
443  #Step function so that only upstream turbines K influence turbine I
444  #Pulse function to account for the width of the wake (check whether
445  #the spanwise separation between I and K is smaller than the half-width
446  #of the wake)
447  wakedeficit += ( (1.0-np.sqrt(1.0-turbk.Ct))
448  /(1+2*kwake*delta_str/turbk.D)**2
449  * step(delta_str)
450  * pulse(delta_span,turbk.D/2.0+kwake*delta_str) )**2
451  #Velocity and Force at turbine i
452  S[i] = Sinf*(1-np.sqrt(wakedeficit))
453  Ftu[i] = 0.5* turbi.Ct * turbi.rotorarea * S[i]**2 * e_str[0]
454  Ftv[i] = 0.5* turbi.Ct * turbi.rotorarea * S[i]**2 * e_str[1]
455  return Ftu,Ftv,S
