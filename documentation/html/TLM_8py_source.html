00972             A[index+Nx,index] += -self.abl.fc
00973             A[index+Nx,index+1*Nx] += -1j*sigma1
00974             A[index+Nx,index+1*Nx] += -2*self.abl.C1*self.abl.dS12/self.abl.H1
00975             A[index+Nx,index+1*Nx] += -2*self.abl.C0*self.abl.S1/self.abl.H1
00976             A[index+Nx,index+3*Nx] += 2*self.abl.C1*self.abl.dS12/self.abl.H1
00977     
00978             #u2 equation
00979             A[index+2*Nx,index] += 1j*k*PHI*self.abl.H1/self.abl.U1
00980             A[index+2*Nx,index] += 2*self.abl.C1*self.abl.dS12/self.abl.H2
00981             A[index+2*Nx,index+2*Nx] += -1j*sigma2
00982             A[index+2*Nx,index+2*Nx] +=  1j*k*PHI*self.abl.H2/self.abl.U2
00983             A[index+2*Nx,index+2*Nx] += -2*self.abl.C1*self.abl.dS12/self.abl.H2
00984             A[index+2*Nx,index+3*Nx] += self.abl.fc
00985     
00986             #v2 equation
00987             A[index+3*Nx,index+1*Nx] += 2*self.abl.C1*self.abl.dS12/self.abl.H2
00988             A[index+3*Nx,index+2*Nx] += -self.abl.fc
00989             A[index+3*Nx,index+3*Nx] += -1j*sigma2
00990             A[index+3*Nx,index+3*Nx] += -2*self.abl.C1*self.abl.dS12/self.abl.H2
00991         
00992         #A matrix convolution - Part 1: Linearisation Wind farm drag
00993         #It is implicitly assumed that a constant forcing model
00994         #with a CTr field is used. 
00995         A[:Nx,:Nx] += -2*self.abl.S1/self.abl.H1*cconv_1D(self.r2c(self.forcing.CTr))
00996         A[Nx:2*Nx,Nx:2*Nx] += -2*self.abl.S1/self.abl.H1*cconv_1D(self.r2c(self.forcing.CTr))
00997 
00998         #A matrix convolution - Part 2: Fringe region forcing
00999         if self.forcing.Lfringe:
01000             A[:Nx,:Nx] += -cconv_1D(self.r2c(self.forcing.Cfr))
01001             A[1*Nx:2*Nx,1*Nx:2*Nx] += -cconv_1D(self.r2c(self.forcing.Cfr))
01002             A[2*Nx:3*Nx,2*Nx:3*Nx] += -cconv_1D(self.r2c(self.forcing.Cfr))
01003             A[3*Nx:4*Nx,3*Nx:4*Nx] += -cconv_1D(self.r2c(self.forcing.Cfr))
01004     
01005         #Set defunct modes to zero
01006         A[0,:]    = 0.
01007         A[Nx,:]   = 0.
01008         A[2*Nx,:] = 0.
01009         A[3*Nx,:] = 0.
01010         A[0,0]       = 1.
01011         A[Nx,Nx]     = 1.
01012         A[2*Nx,2*Nx] = 1.
01013         A[3*Nx,3*Nx] = 1.
01014         return A

01806                 -2*self.abl.C1*self.abl.dS12/self.abl.H1

01827         Axv2 += (-1j*sigma2
01828                 -2*self.abl.C1*self.abl.dS12/self.abl.H2)*v2

01834         Axp1 += sigma1*p1
01835         #p2 equation
01836         #General case

01838         Axp2 += np.ravel(self.PHI)*self.abl.H2*ls*v2
01839         Axp2 += sigma2*p2

01841         indices_klzero = int(Nx/2)*Ny+int(Ny/2)
01842         Axp1[indices_klzero] = p1[indices_klzero]
01843         Axp2[indices_klzero] = p2[indices_klzero]

01846         u1r = self.c2r(u1.reshape(self.grid.shape))

01849         #Convert back to fourier space, cast into 1D array and
01850         #divide by H1 (TLM solves height-averaged equations)
01851         Axu1 += -np.ravel(self.r2c(F1u))/self.abl.H1
01852         Axv1 += -np.ravel(self.r2c(F1v))/self.abl.H1

01854         #Set defunct modes to zero
01855         defunct_k = [i for i in range(Ny)]
01856         defunct_l = [i*Ny for i in range(1,Nx)]
01857         defunctindices = np.array(defunct_k + defunct_l)
01858         Axu1[defunctindices] = 0.
01859         Axv1[defunctindices] = 0.
01860         Axu2[defunctindices] = 0.
01861         Axv2[defunctindices] = 0.
01862         Axp1[defunctindices] = 0.
01863         Axp2[defunctindices] = 0.

02192     def analytic_cubic(self,**kwargs):
02193         #ABL state based on analytical formulas for u and v (Nieuwstadt 1983)
02194         arguments = ['dth','fc','N','G','alpha','kappa','utau','h','H1']
02195         assert all([i in kwargs for i in arguments]), 'Error: some arguments for analytic_quadratic ABL definition are missing'
02196         
02197         self.__gprime = 9.81*kwargs['dth']/288.15
02198         self.__fc = kwargs['fc']
02199         self.__N  = kwargs['N']
02200         self.__U3 = kwargs['G']*np.cos(kwargs['alpha'])
02201         self.__V3 = kwargs['G']*np.sin(kwargs['alpha'])
02202         self.__H1 = kwargs['H1']
02203         self.__H2 = kwargs['h']-self.H1
02204 
02205         h = kwargs['h']
02206         kappa = kwargs['kappa']
02207         utau = kwargs['utau']
02208 
02209         #Velocity deficit vector
02210         Nz = 100
02211         self.__zst = np.linspace(0.,h,Nz)
02212         self.__zs = (self.zst[0:-1]+self.zst[1:])/2.0
02213         C = h*self.fc/kappa/utau
02214         alpha = 0.5+0.5*np.sqrt(1+4j*C)
02215         wd = np.zeros((Nz-1),dtype=np.complex128)
02216         sigma = np.zeros((Nz-1),dtype=np.complex128)
02217         for k in range(Nz-1):
02218             wd[k] = (1j*alpha**2*(mpmath.gamma(alpha))**2)/(kappa*C*mpmath.gamma(2*alpha))*(1-self.zs[k]/h)**(alpha-1)*mpmath.hyp2f1(alpha+1,alpha-1,2*alpha,1-self.zs[k]/h)
02219             sigma[k] = (alpha*(mpmath.gamma(alpha))**2)/(mpmath.gamma(2*alpha))*(1-self.zs[k]/h)**(alpha)*mpmath.hyp2f1(alpha-1,alpha,2*alpha,1-self.zs[k]/h)
02220 
02221         self.__us = self.U3+np.real(wd)*utau
02222         self.__vs = self.V3+np.imag(wd)*utau
02223         taux = np.real(sigma)*utau**2
02224         tauy = np.imag(sigma)*utau**2
02225         tau = np.sqrt(taux**2+tauy**2)
02226         #Compute height averaged quantities
02227         #Layer 1
02228         i1 = np.max(np.where(self.zst<=self.H1))-1
02229         self.__U1 = mypy.trapzst(self.us,self.zst,0,i1)/self.H1
02230         self.__V1 = mypy.trapzst(self.vs,self.zst,0,i1)/self.H1
02231         self.__nu1 = mypy.trapzst(kappa*utau*self.zs*(1-self.zs/h)**2,self.zst,0,i1)/self.H1
02232         tau01 = utau**2
02233         self.__C0 = tau01/self.S1**2
02234         #Layer 2
02235         self.__U2 = mypy.trapzst(self.us,self.zst,i1+1,Nz)/self.H2
02236         self.__V2 = mypy.trapzst(self.vs,self.zst,i1+1,Nz)/self.H2
02237         self.__nu2 = mypy.trapzst(kappa*utau*self.zs*(1-self.zs/h)**2,self.zst,i1+1,Nz)/self.H2
02238         tau12 = tau[i1+1]
02239         self.__C1 = tau12/self.dS12**2
02240         self.__C2 = 0.

02242     def ERA5(self,**kwargs):
02243         arguments = ['H1','T2','blh','ust','wth','phi',
02244                      'zs','us','vs','ths','Gmode']
02245         assert all([i in kwargs for i in arguments]),'Error: some arguments for ERA5 based ABL definition are missing'
02246 
02247         gravity = 9.80665    # [m s-2]
02248         P0 = 1.e5 # Reference pressure [Pa]
02249         R_air = 287.058 # Specific gas constant for dry air [J kg-1 K-1]
02250         Cp_air = 1005   # Specific heat of air [J kg-1 K-1]
02251         kappa  = 0.41   # Von Karman constant
02252         eps = 0.609133  # Rv/Rd-1
02253         omega = 7.2921159e-5    # angular speed of the Earth [rad/s]
02254 
02255         #Surface parameters
02256         ust = kwargs['ust']
02257         wth = kwargs['wth']
02258         T2  = kwargs['T2']
02259         blh = kwargs['blh']
02260         zeta = -2.0*kappa*gravity*wth/(T2*ust**3)
02261 
02262         #Vertical profiles (arrays are in reversed order)
02263         self.__zs  = kwargs['zs'][::-1]
02264         self.__us  = kwargs['us'][::-1]
02265         self.__vs  = kwargs['vs'][::-1]
02266         self.__ths = kwargs['ths'][::-1]
02267 
02268         #Estimate inversion parameters
02269         zCI = self.zs[self.zs<5000]
02270         thCI = self.ths[self.zs<5000]
02271         if 'dh_max' in kwargs:
02272             dh_max = kwargs['dh_max']
02273         else:
02274             dh_max=None
02275 
02276         if zeta>0.02:
02277             #Ignore temperature decrease inside SBL
02278             #(we are trying to indentify the mixing layer that preceded this SBL)
02279             thCI[zCI<blh] = interpolate.interp1d(zCI,thCI)(blh)
02280             CIestimate = CIops.RZfit(zCI,thCI,p0=[0.9,0.1,T2,1000.,100.0],
02281                                             dh_max=dh_max)
02282         else:
02283             #Ignore temperature increase in CBL surface layer
02284             thCI[0:np.argmin(thCI)] = np.min(thCI)
02285             CIestimate = CIops.RZfit(zCI,thCI,p0=[0.9,0.1,T2,blh,100.0],
02286                                             dh_max=dh_max)
02287 
02288         #No inversion strength in the following cases:
02289         #a<=0.2: encroachment
02290         #a<=2*b: inversion lapse rate is equal to or smaller than free lapse rate
02291         if (CIestimate['a']<=0.2 or CIestimate['a']<=2*CIestimate['b']):
02292             H = np.max([CIestimate['h1'],kwargs['H1']+10])
02293             self.__gprime = 0.
02294         else:
02295             H = np.max([CIestimate['h1'],kwargs['H1']+10])
02296             self.__gprime = gravity*CIestimate['dth']/T2
02297 
02298         #Flux profile
02299         tau    = np.zeros(self.zs.shape)
02300         nu     = np.zeros(self.zs.shape)
02301         if zeta>0.0:
02302             tau[self.zs<=blh] = ust**2*(1-self.zs[self.zs<=blh]/blh)**(1.5)
02303             nu[self.zs<=blh]  = kappa*ust*self.zs[self.zs<=blh]*(1-self.zs[self.zs<=blh]/blh)**2
02304         else:
02305             tau[self.zs<=H] = ust**2*(1-self.zs[self.zs<=H]/H)
02306             nu[self.zs<=H]  = kappa*ust*self.zs[self.zs<=H]*(1-self.zs[self.zs<=H]/H)**2
02307         self.__taus = tau
02308 
02309         fu = interpolate.interp1d(self.zs,self.us,fill_value='extrapolate')
02310         fv = interpolate.interp1d(self.zs,self.vs,fill_value='extrapolate')
02311         ft = interpolate.interp1d(self.zs,self.taus,fill_value='extrapolate')
02312         fn = interpolate.interp1d(self.zs,nu,fill_value='extrapolate')
02313         #Compute height averaged quantities
02314         #Layer 1
02315         self.__H1 = kwargs['H1']
02316         z = np.linspace(0,self.H1,100)
02317         self.__U1 = np.trapz(fu(z),z)/self.H1
02318         self.__V1 = np.trapz(fv(z),z)/self.H1
02319 #        self.__nu1 = np.trapz(fn(z),z)/self.H1
02320         #Layer 2
02321         self.__H2 = H-self.H1
02322         z = np.linspace(self.H1,self.H,100)
02323         self.__U2 = np.trapz(fu(z),z)/self.H2
02324         self.__V2 = np.trapz(fv(z),z)/self.H2
02325 #        self.__nu2 = np.trapz(fn(z),z)/self.H2
02326         #Layer 3
02327         if kwargs['Gmode'] == 'h1':
02328             self.__U3 = np.asscalar(fu(CIestimate['h1']))
02329             self.__V3 = np.asscalar(fv(CIestimate['h1']))
02330         elif kwargs['Gmode'] == 'h2':
02331             self.__U3 = np.asscalar(fu(CIestimate['h2']))
02332             self.__V3 = np.asscalar(fv(CIestimate['h2']))
02333         elif kwargs['Gmode'] == 'top':
02334             self.__U3 = np.asscalar(fu(5000.))
02335             self.__V3 = np.asscalar(fv(5000.))
02336         elif kwargs['Gmode'] == 'avg':
02337             z = np.linspace(self.H,5000.,1000)
02338             self.__U3 = np.trapz(fu(z),z)/(z[-1]-z[0])
02339             self.__V3 = np.trapz(fu(z),z)/(z[-1]-z[0])
02340         else:
02341             print('Gmode unknown, abort')
02342             return 1
02343 
02344         #Stress levels
02345         tau01 = ust**2
02346         tau12 = ft(self.H1)
02347         tau23 = ft(self.H)
02348         self.__C0 = tau01/self.S1**2
02349         self.__C1 = tau12/self.dS12**2
02350         self.__C2 = tau23/self.dS23**2
02351 
02352         #Turbulent intensity at hub height
02353         if zeta>0.0:
02354             #From Nieuwstadt (1984): q/sqrt(tau) = 3
02355             tke = 4.5*tau
02356             f = interpolate.interp1d(self.zs,np.sqrt(2./3.*tke/self.Ms))
02357         else:
02358             #From Stull (1988): q^2/tau = 8.5+2.5
02359             tke = 5.5*tau
02360             f = interpolate.interp1d(self.zs,np.sqrt(2./3.*tke/self.Ms))
02361         self.__TI = np.asscalar(f(self.H1/2.0))
02362         self.__TIs = f(self.zs)
02363     
02364         #Other ABL parameters
02365         self.__N  = np.sqrt(gravity*CIestimate['gamma']/T2)
02366         self.__fc = 2*omega*np.sin(kwargs['phi'])

02672 def cconv_1D(b,method='Fast'):
02673     '''
02674     Compose a matrix that expresses the 1D circular convolution
02675     as a matrix vector multiplication (conv(a,b) = Ba)
02676     
02677     b is assumed to be a vector with wave numbers centered around 0
02678     (the 0 mode corresponds to b[int(b.shape[0]/2)] )
02679     '''
02680     if method=='Naive':
02681         #Naive method
02682         N = b.shape[0]
02683         B = np.zeros((N,N))
02684         for k in range(N):
02685             for k2 in range(N):

02687                 if kshifted>=N:
02688                     kshifted -= N
02689                 B[k,k2] += b[kshifted]

02698 def cconv_2D(b,method='Fast'):
02699     '''
02700     Compose a matrix that expresses the 2D circular convolution
02701     as a matrix vector multiplication (conv(a,b) = Ba)
02702     
02703     b is assumed to be a matrix with wave numbers centered around 0
02704     (the 0 mode corresponds to b[int(b.shape[0]/2),int(b.shape[1]/2] )
02705     '''
02706     Nk = b.shape[0]
02707     Nl = b.shape[1]
02708     if method=='Naive':
02709         #Naive method
02710         B = np.zeros((Nk*Nl,Nk*Nl),dtype=np.complex128)
02711         for k in range(Nk):
02712             for l in range(Nl):
02713                 index = l + Nl*k
02714                 for k2 in range(Nk):
02715                     for l2 in range(Nl):
02716                         kshifted = k-k2+int(Nk/2)
02717                         lshifted = l-l2+int(Nl/2)
02718                         if kshifted>=Nk:
02719                             kshifted -= Nk
02720                         if lshifted>=Nl:
02721                             lshifted -= Nl
02722                         index2 = l2 + Nl*k2
02723                         B[index,index2] += b[kshifted,lshifted]
02724 
02725     elif method=='Fast':
02726         #Fast method
02727         bshift = np.fft.ifftshift(b)
02728         #Construct an array with the Nk different block toeplitz matrices
02729         #(undo the fftshift on the 1D input vector as this is also done in cconv_1D)
02730         c = np.array([cconv_1D(np.fft.fftshift(bshift[i,:])) for i in range(Nk)])
02731         #Construct amtrix with toeplitz structure,
02732         #but inputs are now block matrices instead of scalars
02733         #cpied from implementation of scipy.linalg.toeplitz)
02734         vals = np.concatenate((c[1:,:,:],c))

02738         x = np.swapaxes(A,1,2).reshape(Nk,Nl,Nk*Nl)
02739         B = x.reshape(Nk*Nl,Nk*Nl)
