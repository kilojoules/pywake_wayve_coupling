1334  def continuity(self,u1c,v1c,u2c,v2c,p1c,p2c):

1350  #Ks, Ls = np.meshgrid(self.grid.ks,self.grid.ls,indexing='ij')

1358  ##because it does not appear in the system of equations (the continuity

1363  ##There is no issue for values of sigma1,2 close but not equal to zero
1364  #eta1[sigma1==0.]=p1c[sigma1==0.]/self.PHI[sigma1==0.]
1365  #eta2[sigma2==0.]=p2c[sigma2==0.]/self.PHI[sigma2==0.]
1366  #eta = eta1+eta2

1369  #Eta can also be found by the relation p1,2=PHI*eta1,2
1370  #The result is identical to the displacement found with the continuity

1374  #two very small numbers (order of 1.0e-21) which is inaccurate
1375  with np.errstate(divide='ignore',invalid='ignore'):

1387  eta1[(self.PHI==0.) & (np.isclose(sigma1,0.,atol=1.e-8))] = 0.

1592  def Ax(self,x):
1593  '''
1594  Compute the matrix vector product A*x
1595 
1596  The pressure is split in two variables p1,2=Phi*eta_1,2 which are treated
1597  as independent variables instead of using a direct substituion in terms of
1598  eta_1,2 in the momentum equations. The reason is that for cases where
1599  sigma_1,2 is zero but l or k is not, the pressure is not zero but follows
1600  indirectly from the continuity equation. When k=l=0 the continuity
1601  equations become trivial and should be replaced by p1,2=0
1602 
1603  Parameters
1604  ----------
1605  x: 1d numpy array
1606  input vector
1607 
1608  Returns
1609  -------
1610  _: 1d numpy array
1611  matrix vector product A*x
1612  '''
1613  N = self.grid.N
1614  Nx = self.grid.Nx
1615  Ny = self.grid.Ny
1616  #Initialise components of the matrix vector product
1617  Axu1 = np.zeros((N),dtype=np.complex128)
1618  Axv1 = np.zeros((N),dtype=np.complex128)
1619  Axu2 = np.zeros((N),dtype=np.complex128)
1620  Axv2 = np.zeros((N),dtype=np.complex128)
1621  Axp1 = np.zeros((N),dtype=np.complex128)
1622  Axp2 = np.zeros((N),dtype=np.complex128)
1623  #Extract dependent variables (u1,v1,u2,v2,p1,p2) from the vector
1624  u1 = x[0:N]
1625  v1 = x[N:2*N]
1626  u2 = x[2*N:3*N]
1627  v2 = x[3*N:4*N]
1628  p1 = x[4*N:5*N]
1629  p2 = x[5*N:6*N]
1630  #Create some additional vectors
1631  Ks, Ls = np.meshgrid(self.grid.ks,self.grid.ls,indexing='ij')
1632  ks = np.ravel(Ks)
1633  ls = np.ravel(Ls)
1634  sigma1 = self.abl.U1*ks+self.abl.V1*ls
1635  sigma2 = self.abl.U2*ks+self.abl.V2*ls
1636  #Regular entries
1637  #u1 equation
1638  Axu1 += -1j*sigma1*u1
1639  Axu1 += -1j*ks*(p1+p2)
1640  Axu1 += +self.abl.fc*v1
1641  Axu1 += -self.abl.C0*( (self.abl.S1**2+self.abl.U1**2)/
1642  (self.abl.S1*self.abl.H1) )*u1
1643  Axu1 += -self.abl.C0*( (self.abl.U1*self.abl.V1)/
1644  (self.abl.S1*self.abl.H1) )*v1
1645  Axu1 += -self.abl.C1*( (self.abl.dS12**2+self.abl.dU12**2)/
1646  (self.abl.dS12*self.abl.H1) )*(u1-u2)
1647  Axu1 += -self.abl.C1*( (self.abl.dU12*self.abl.dV12)/
1648  (self.abl.dS12*self.abl.H1) )*(v1-v2)
1649  Axu1 += -self.abl.nu1*(ks**2+ls**2)*u1
1650  #v1 equation
1651  Axv1 += -1j*sigma1*v1
1652  Axv1 += -1j*ls*(p1+p2)
1653  Axv1 += -self.abl.fc*u1
