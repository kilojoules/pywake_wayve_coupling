286  def Mx(self,x):
287  '''
288  Compute the matrix vector product M*x where preconditioner M = inv(P)
289  approximates inv(A) (in Smith model, the approximation is exact)
290  The matrix vector product Mx is found by solving Py=x
291 
292  Parameters
293  ----------
294  x: 1d numpy array
295  input vectori
296 
297  Returns
298  -------
299  _: 1d numpy array
300  matrix vector product M*x
301  '''
302  N = self.grid.N
303  M = np.zeros((2,N),dtype=np.complex128)
304  X = x.reshape(2,N)
305  for index, k in enumerate(self.grid.ks):
306  P = np.zeros((2,2),dtype=np.complex128)
307  sigma1 = self.abl.U1*k
308  #u1 equation
309  P[0,0] = (-1j*sigma1-self.abl.C0-self.abl.C1
310  +1j*k*self.PHI[index]*self.abl.H1/self.abl.U1)
311  #P[0,1] = 0.
312  #v1 equation
313  #P[1,0] = 0.
314  P[1,1] = (-1j*sigma1-self.abl.C0-self.abl.C1)
315  M[:,index] = scipy.linalg.solve(P,X[:,index])
316  #Defunct mode
317  M[:,0] = 0.
318  return M.reshape(2*N)

432  '''
433  Compute boundar-layer displacement based on given velocity field
434 
435  Parameters
436  ----------
437  u1c,v1c,pc: 2d numpy array
438  perturbation velocities and pressure (Fourier space)
439 
440  Returns
441  -------
442  eta: 2d numpy array
443  boundary-layer displacement
444  '''
445  ##Using the continuity equation
446  #Ks, Ls = np.meshgrid(self.grid.ks,self.grid.ls,indexing='ij')
447  #sigma1 = self.abl.U1*Ks+self.abl.V1*Ls
448  #with np.errstate(divide='ignore',invalid='ignore'):
449  # eta = -self.abl.H1/sigma1*(Ks*u1c+Ls*v1c)

454  ##The value of eta is replaced with the limit value p/PHI to make

457  ##There is no issue for values of sigma1 close but not equal to zero

461  #Eta can also be found by the relation p=PHI*eta
462  #The result is identical to the displacement found with the continuity

464  #However, this method is numerically more stable as it only involves a

467  #The only issue here is when PHI=0., which can occur for gprime=0.
468  eta = pc/self.PHI
