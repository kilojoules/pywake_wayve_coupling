49  def LESbased(self,**kwargs):
50  '''
51  Method to specify the forcing parameters based on LES data
52 
53  This routine assimes that the LES data is obtained with SP-Wind, and
54  the input parameters include a specific data structure of the SP-Wind
55  post-processing tools
56 
57  Parameters
58  ----------
59  sim: Simulation object (defined in simulation.py of py4sp package)
60  LES simulation meta data structure
61  abl: ABL object (defined in TLM.py or SLM.py)
62  atmospheric state of the Three-layer model
63  '''
64  assert all([i in kwargs for i in ['sim','abl']]), 'Error: some arguments for LESbased forcing definition are missing'
65 
66  sim = kwargs['sim']
67  abl = kwargs['abl']
68  if 'efficiency' in kwargs:
69  efficiency = kwargs['efficiency']
70  else:
71  efficiency=1.0
72 
73  CD = (16.0*sim.wf.Ctp)/(4.0+sim.wf.Ctp)**2
74  diameter = 2.0*sim.wf.r[0]
75  rotorarea = np.pi*sim.wf.r[0]**2
76  n = 1.0/(sim.wf.sx*sim.wf.sy*diameter**2)
77  areas = mypy.disk_areas(sim.grid.zst,sim.wf.z[0],sim.wf.r[0])
78  udisk = np.sum(abl.Ms*areas)/rotorarea
79  FT = 0.5*efficiency*CD*udisk**2*rotorarea
80  self.__CT = n*FT/abl.S1**2
81  self.__length = np.max(sim.wf.x)-np.min(sim.wf.x)+diameter*sim.wf.sx
82  self.__width = np.max(sim.wf.y)-np.min(sim.wf.y)+diameter*sim.wf.sy

501  '''
502  Compute zero-th order wind-farm force on specified grid
503 
504  Parameters
505  ----------
506  abl: ABL object (defined in TLM.py)
507  atmospheric state
508  grid: Grid object (defined in TLM.py)
509  numerical grid
510 
511  Returns

528  def F1(self,abl,grid,u1r,v1r):
529  '''
530  Compute first order wind-farm force on specified grid for given
531  perturbation velocities
532 
533  Parameters
534  ----------
535  abl: ABL object (defined in TLM.py)
536  atmospheric state
537  grid: Grid object (defined in TLM.py)
538  numerical grid
539  u1r, v1r: 2d numpy array
540  velocity perturbation fields (in real space)
541 
542  Returns
543  -------
544  F1u,F1v: 2d numpy array
545  first order wind-farm force in x and y (same shape as grid)
546  '''
547  #Get velocity perturbations
548  function = getattr(self,'u_'+self.coupling)
549  u1inf,v1inf = function(abl,grid,u1r,v1r)
550  #Unit vectors and their derivatives
551  e_str = WakeModel.e_streamwise(abl.U1,abl.V1)
552  E_str = WakeModel.e_str_jac(abl.U1,abl.V1)
553  #Filter turbine forces onto grid
554  F1u = np.zeros(grid.shape)
555  F1v = np.zeros(grid.shape)
556  for index,turb in enumerate(self.turbines):
557  F1u += (0.5 * turb.Ct * turb.rotorarea *
558  ( 2*self.St[index]*self.Stjac[index,0]*e_str[0] +
559  self.St[index]**2*E_str[0,0] ) *
560  u1inf*turb.footprint(grid,self.Lfilter) )
561  F1u += (0.5 * turb.Ct * turb.rotorarea *
562  ( 2*self.St[index]*self.Stjac[index,1]*e_str[0] +
563  self.St[index]**2*E_str[0,1] ) *
564  v1inf*turb.footprint(grid,self.Lfilter) )
565  F1v += (0.5 * turb.Ct * turb.rotorarea *
566  ( 2*self.St[index]*self.Stjac[index,0]*e_str[1] +
567  self.St[index]**2*E_str[1,0] ) *
568  u1inf*turb.footprint(grid,self.Lfilter) )
569  F1v += (0.5 * turb.Ct * turb.rotorarea *
570  ( 2*self.St[index]*self.Stjac[index,1]*e_str[1] +
571  self.St[index]**2*E_str[1,1] ) *
572  v1inf*turb.footprint(grid,self.Lfilter) )
573  return F1u,F1v
